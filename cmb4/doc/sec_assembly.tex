\section{Assembly of the power spectrum}
\label{sec:assembly}
This section turns the map of Secs.~\ref{sec:observable}, \ref{sec:map2}, \ref{sec:map3} into explicit kernels
$K_1$, $K_2$ (and, when available, $K_{3c}$) and assembles $C^{(11)}_\ell$, $C^{(22)}_\ell$, $C^{(13)}_\ell$ with the rules of
Sec.~\ref{sec:harmonic}.  Every statement below is either derived in the text or quoted with the check that established it
(App.~\ref{app:ledger}; code in App.~\ref{app:repro}).

\paragraph{Conventions of this section.}  Observer label at the origin, ray $\bar\vx(\eta)=\nhat\chi$, $\chi=\eta_o-\eta$
(Sec.~\ref{sec:setup}; $\nhat$ = observation direction), so a field on the ray carries $e^{+i\vk\cdot\nhat\chi}$.  Rayleigh:
$e^{i\vk\cdot\nhat\chi}=4\pi\sum_{\ell m}i^\ell j_\ell(k\chi)Y_{\ell m}(\nhat)Y^*_{\ell m}(\khat)$, and
$(i\nhat\cdot\khat)^s e^{i\vk\cdot\nhat\chi}\to4\pi i^\ell j^{(s)}_\ell(k\chi)Y^*_{\ell m}(\khat)$, the convention of
Sec.~\ref{sec:harmonic}.  We write
$\mu_a\equiv\nhat\cdot\khat_a$, $E_a\equiv e^{ik_a\mu_a\chi}$, and $\dot T\equiv\partial_\eta T$.

\subsection{First-order kernel, sky monopole and observer objects}
\label{sec:asm-first}
From \eqref{eq:L1} with $\Phi_1=\Psi_1=\zeta\TP$, $\delta_{T1}=\zeta T_{\delta_T}$, $\vv_1=i\vk\,T_V\zeta$ \eqref{eq:transfer},
\begin{equation}
K_1(\vk;\nhat)=\big[T_{\delta_T}+\TP-ik\mu T_V\big]_{\eta_*}E(\eta_*)-\big[T_{\delta_T}+\TP-ik\mu T_V\big]_{\eta_o}
+\int_{\eta_*}^{\eta_o}\!\dd\eta\,2\dot T_\Phi\,E(\eta),\qquad \mu=\nhat\cdot\khat,
\label{eq:asm-K1}
\end{equation}
including the observer monopole and dipole.  Its multipoles are
\begin{equation}
K_{1,\ell m}(\vk)=4\pi\,i^\ell\,\Delta_\ell(k)\,Y^*_{\ell m}(\khat),
\label{eq:asm-K1lm}
\end{equation}
with the real transfer function $\Delta_\ell(k)$ of Proposition~\ref{prop:Delta} (closed form of the ISW part:
Theorem~\ref{thm:isw}); the observer terms are $-\delta_{\ell0}(T_{\delta_T}+\TP)(k,\eta_o)+\tfrac13\delta_{\ell1}kT_V(k,\eta_o)$.
($K_1$ rebuilt from $\Delta_\ell$, $\ell\le40$, equals the numerical line-of-sight kernel to $7\times10^{-15}$; the sign of the
observer dipole is fixed by the comparison with the non-perturbative planar solver of Sec.~\ref{sec:check_endtoend}, $a_{10}[\Theta_1]=2.0812147132$ in both.)
The sky monopole entering \eqref{eq:theta_orders} is the $\ell=0$ projection,
\begin{equation}
\sky{m_1}=\int\!\frac{\dd^3k}{(2\pi)^3}\,\zeta_\vk\,\Delta_0(k),\qquad
\big\langle\sky{m_1}^2\big\rangle=\frac1{2\pi^2}\int k^2\dd k\,P(k)\,\Delta_0(k)^2=\frac{C^{(11)}_0}{4\pi},
\label{eq:asm-monopole}
\end{equation}
with $\Delta_0=O(k^4)$ as $k\to0$ (Sec.~\ref{sec:linear}).

\paragraph{Observer clock.}  The observer measures at fixed proper time $\Delta\tau=\int_{\eta_{\rm init}}^{\eta_o}a\,\dd\eta/\hat u^0$
from the initial-data surface (Sec.~\ref{sec:observer}, \eqref{eq:clock}; second order in \eqref{eq:deta_o2}).  At first order $1/\hat u^0=1+\Psi_1$ on the worldline,
so $\Delta\tau=\int\eta(1+\eps\Psi_1)\dd\eta$; holding it fixed gives $\bar\eta_o\,\delta\eta_{o1}+\int_0^{\bar\eta_o}\eta\Psi_1(\eta,\mathbf0)\dd\eta=0$
up to an initial-surface term $O(\eta_i^2)\to0$.  Per mode ($\delta\eta_{o1}=\int\frac{\dd^3k}{(2\pi)^3}\zeta_\vk\,\delta\eta_o(k)$), using
$\int x\TP(x)\dd x=2\sin x/x$,
\begin{equation}
\delta\eta_o(k)=-\frac1{\eta_o}\int_0^{\eta_o}\!\eta\,\TP(k,\eta)\,\dd\eta=\frac{6}{k^2\eta_o}\Big(1-\frac{\sin x_o}{x_o}\Big)
\ \xrightarrow{k\to0}\ \frac{\eta_o}3,\qquad x_o\equiv\frac{k\eta_o}{\sqrt3}.
\label{eq:asm-clock}
\end{equation}
The clock shift is the time integral of the \emph{lapse} $\Psi_1$, not of $\delta_{T1}$ (the two differ for $k\eta_o\sim1$).
The limit $\eta_o/3$ is the constant-mode value $\eta_o=\bar\eta_o/\lambda$, $\lambda=1-c/3+O(c^2)$, of the exact dilation family
(Sec.~\ref{sec:tadpole}, \eqref{eq:dilation}).

\paragraph{Observer displacement.}  The tagged worldline (Sec.~\ref{sec:observer}; label = position as $\eta\to0$) moves by
$\delta\vx_{o1}=\int_0^{\eta_o}\vv_1(\eta,\mathbf0)\dd\eta=\int\frac{\dd^3k}{(2\pi)^3}\zeta_\vk\,i\vk\,X_o(k)$ with
\begin{equation}
X_o(k)=\int_0^{\eta_o}T_V(k,\eta)\,\dd\eta=\frac3{k^2}\Big(\frac{2\sin x_o}{x_o}-\cos x_o-1\Big)\ \xrightarrow{k\to0}\ \frac{\eta_o^2}6 .
\label{eq:asm-disp}
\end{equation}
\paragraph{Emission-surface shift.}  From $1+\delta_T(\eta_e,\vx(\eta_e))=\eta_e/\eta_*$ \eqref{eq:emission},
\begin{equation}
\delta\eta_{e1}=\eta_*\,\delta_{T1}(\eta_*,\bar\vx_*)\quad\Rightarrow\quad \delta\eta_e(\vk)=\eta_*\,T_{\delta_T}(k,\eta_*)\,E(\eta_*).
\label{eq:asm-emission}
\end{equation}
At first order $\delta\eta_{o1}$ and $\delta\vx_{o1}$ do not enter $\Theta^{(1)}$ ($S$ has no background part); they first appear in
$K_2$.

\subsection{The second-order kernel \texorpdfstring{$K_2$}{K2}}
\label{sec:asm-K2}
\paragraph{Factor bookkeeping.}  With $\Lambda=-S=\eps L_1+\tfrac{\eps^2}2L_2+\dots$ \eqref{eq:mdef}, the term list of
Sec.~\ref{sec:map2} is $L_2/2$; call its kernel $K_{L_2/2}(\vk_1,\vk_2;\nhat)$ (not symmetric).  Then
$m_2=L_2+L_1^2$, and by \eqref{eq:theta_orders} $\Theta^{(2)}_{\ell m}=m_{2,\ell m}-2\sky{m_1}m_{1,\ell m}$ for $\ell\ge1$.  With
$\Theta=\eps\Theta^{(1)}+\tfrac{\eps^2}2\Theta^{(2)}$ and $\Theta^{(2)}=\int\!\!\int\zeta\zeta K_2$ \eqref{eq:Kr}, the symmetric kernel is
\begin{equation}
\boxed{\;K_2(\vk_1,\vk_2;\nhat)=2\,{\rm Sym}\big[K_{L_2/2}\big]+K_1(\vk_1)K_1(\vk_2)-\Delta_0(k_1)K_1(\vk_2)-\Delta_0(k_2)K_1(\vk_1)\;}
\qquad(\ell\ge1),
\label{eq:asm-K2}
\end{equation}
${\rm Sym}[f]=\tfrac12[f(\vk_1,\vk_2)+f(\vk_2,\vk_1)]$; the factor 2 converts $L_2/2$ into $L_2$, the product $K_1K_1$ is the $L_1^2$ of $m_2$,
and $-2\sky{m_1}m_1$ symmetrised gives the last two terms (with \eqref{eq:asm-monopole}).  The $\ell=0$ part of \eqref{eq:asm-K2} is
not the monopole of $\Theta^{(2)}$ (which vanishes) and is never used.  We use the \emph{direct} (redshift) form \eqref{eq:L2direct} of $L_2$, which equals
the shear form modulo the conservation equations; the two give the same $K_2$ to $6\times10^{-15}$ at generic momenta
(Sec.~\ref{sec:asm-cl22}).

\paragraph{Fourier rules and named objects.}  A first-order field $X_1$ on the ray contributes, in slot $a\in\{1,2\}$,
$\zeta_{\vk_a}T_X(k_a,\eta)E_a$, with $\partial_j\to ik_{a,j}$, $\nhat\cdot\vv_1\to ik_a\mu_aT_V$.  Geometry:
$\mu\equiv\khat_1\cdot\khat_2$, $\vK\equiv\vk_1+\vk_2$, $K_n\equiv\nhat\cdot\vK=k_1\mu_1+k_2\mu_2$,
$P_\perp\equiv\vk_{1\perp}\cdot\vk_{2\perp}=k_1k_2(\mu-\mu_1\mu_2)$ ($\vk_\perp=\vk-\nhat(\nhat\cdot\vk)$), $E_K\equiv E_1E_2=e^{iK_n\chi}$.
Two first-order combinations recur: the direct-form line-of-sight integrand and boundary function,
\begin{align}
J_a&\equiv2ik_a\mu_a\TP(k_a,\eta)E_a\ \ (=2\partial_n\Psi_1=-\Gamma^0_{kk}|_1),\nonumber\\
U_a&\equiv\big[\TP+ik_a\mu_aT_V-T_{\delta_T}\big](k_a,\eta)\,E_a\ \ (=\Psi_1+\nhat\cdot\vv_1-\delta_{T1}),
\label{eq:asm-JU}
\end{align}
the kernels of $\mathcal G_1$ and $B_1$ of \eqref{eq:L2direct}, so that $L_1=\int_{\eta_*}^{\eta_o}J\,\dd\eta+[U]^{o}_{e}$, equal to \eqref{eq:L1}
after one integration by parts.  The perturbed photon
direction and position along the ray, $\vk_1(\eta)=\vk_{1}^{\rm init}-\int_\eta^{\eta_o}\mathbf G_1\dd\eta'$ and
$\Delta\vx_1(\eta)=\delta\vx_{o1}+\nhat\,\delta\eta_{o1}-\int_\eta^{\eta_o}\vk_1\dd\eta'$ (\eqref{eq:k1def}, \eqref{eq:Dx1def}; for the sign of the $\mathbf G_1$ integral see Remark~\ref{rem:signslip}), are, per mode $a$,
\begin{align}
\vk_{1\perp}&=i\vk_{a\perp}\,\mathcal F_a(\eta), &
\mathcal F_a(\eta)&\equiv T_V(k_a,\eta_o)+2\int_\eta^{\eta_o}\!\dd\eta'\,\TP(k_a,\eta')E_a(\eta'),\nonumber\\
\Delta\vx_{1\perp}&=i\vk_{a\perp}\,\mathcal D_{\perp,a}(\eta), &
\mathcal D_{\perp,a}(\eta)&\equiv X_o(k_a)-\int_\eta^{\eta_o}\!\dd\eta'\,\mathcal F_a(\eta'),\nonumber\\
\nhat\cdot\Delta\vx_1&=\mathcal D_{r,a}(\eta), &
\mathcal D_{r,a}(\eta)&\equiv ik_a\mu_aX_o(k_a)+\delta\eta_o(k_a)+2\int_\eta^{\eta_o}\!\dd\eta'\,\TP(k_a,\eta')E_a(\eta'),
\label{eq:asm-functionals}
\end{align}
($\mathcal F$: aberration at the observer plus accumulated deflection; $\mathcal D_\perp$: lensing displacement; $\mathcal D_r$: radial
displacement = Shapiro delay + clock + observer displacement), while $\nhat\cdot\vk_1=-(\Phi_1+\Psi_1)$ is the null condition.
A second-order field contributes its pair kernel times $E_K$ (Sec.~\ref{sec:einstein}, \eqref{eq:kernel_def} with $\vq\to\vk_1$,
$\vp\to\vk_2$, $\vk\to\vK$):
\begin{align}
&\Phi_2\equiv\mathcal K_{\Phi_2}(K,k_1,k_2;\eta),\quad \Psi_2\equiv\mathcal K_{\Phi_2}-K_Q,\nonumber\\
&\delta_{T2}\equiv\tfrac14K_{\delta_2}-\tfrac3{16}T_\delta(k_1)T_\delta(k_2),\quad V_2\ (\vv^L_2=\nabla V_2)\ \text{from \eqref{eq:operators}},\nonumber\\
&\nhat\!\cdot\!P(\Khat)\vk_a=k_a\mu_a-\frac{K_n\,\vK\cdot\vk_a}{K^2},\nonumber\\
&\nhat\cdot\vv^T_2=\tfrac12\big[\nhat\!\cdot\!P(\Khat)\vk_1\,K^{(12)}_{v^T}+\nhat\!\cdot\!P(\Khat)\vk_2\,K^{(21)}_{v^T}\big],\quad
\nhat\cdot\boldsymbol\omega_2=\tfrac12\big[\nhat\!\cdot\!P\vk_1\,K^{(12)}_{\omega}+\nhat\!\cdot\!P\vk_2\,K^{(21)}_{\omega}\big],\nonumber\\
&h_{2,nn}=\Big[(\nhat\!\cdot\!P\vk_1)^2-\tfrac12\Big(1-\frac{K_n^2}{K^2}\Big)\Big(k_1^2-\frac{(\vK\cdot\vk_1)^2}{K^2}\Big)\Big]\mathcal H,\qquad
\mathcal H\equiv\mathcal K_h(K,k_1,k_2;\eta),
\label{eq:asm-second}
\end{align}
with $K^{(ab)}_X\equiv K_X(\vk_a,\vk_b)$, $\mathcal K_{\Phi_2}$, $\mathcal K_h$ the declared integrals \eqref{eq:declared2},
$K_Q,K_\omega,K_{v^T}$ from \eqref{eq:second_kernels}, $K_{\delta_2}$ the $00$-constraint kernel (App.~\ref{app:kernels}), and
$\delta_{T2}$ from $1+\delta_T=(1+\delta)^{1/4}$.

\subsection{Channels of \texorpdfstring{$K_2$}{K2}}
\label{sec:asm-channels}
$K_2=\sum_cK^c$ is organised by the harmonic type of Sec.~\ref{sec:harmonic} (B: products of maps at one point; C: single-time
convolution kernels with phase $E_K$; D: nested two-time or endpoint$\times$functional products).  ``Ray'' channels are
$\int_{\eta_*}^{\eta_o}\dd\eta$ of the expression; ``obs'' channels are evaluated at $\eta_o$, $\chi=0$ (so $E_a=1$); ``emi'' channels at
$\eta_*$, $\chi_*$.  $\sum_{(ab)}$ denotes the sum over $(a,b)=(1,2),(2,1)$.  Each compact form below was checked symbolically against
the machine-generated kernel of App.~\ref{app:k2channels} (residual $0$), and each machine kernel against the numerical engine
($\le1.8\times10^{-15}$).

\paragraph{Type B (products and normalisation).}
\begin{align}
K^{\rm B,prod}&=K_1(\vk_1)K_1(\vk_2)-\Delta_0(k_1)K_1(\vk_2)-\Delta_0(k_2)K_1(\vk_1),\nonumber\\
K^{\rm B,obs}_{1\times1}&=\Big[P_\perp T_{V1}T_{V2}-2T_{\Phi1}T_{\Phi2}+T_{\delta_T1}T_{\delta_T2}-2i\big(T_{\Phi1}T_{V2}k_2\mu_2+T_{\Phi2}T_{V1}k_1\mu_1\big)\Big]_{\eta_o},\nonumber\\
K^{\rm B,emi}_{1\times1}&=\Big[E_1E_2\Big(P_\perp T_{V1}T_{V2}+2T_{\Phi1}T_{\Phi2}-T_{\delta_T1}T_{\delta_T2}+2i\big(T_{\Phi1}T_{V2}k_2\mu_2+T_{\Phi2}T_{V1}k_1\mu_1\big)\Big)\Big]_{\eta_*},
\label{eq:asm-chB}
\end{align}
with $T_{Xa}\equiv T_X(k_a,\cdot)$: the quadratic part of the boundary function $\ln(-\hat u\cdot k)-\ln(1+\delta_T)$ at the two
endpoints.  The two brackets are not opposite: the $P_\perp T_{V1}T_{V2}$ ($v_{1\perp}^2$) term has the same sign at both ends.

\paragraph{Type C (single-time second-order fields and ray-local products).}
\begin{align}
K^{\rm C,ray}_{\Phi_2\Psi_2}&=E_K\big[\dot\Phi_2-\dot\Psi_2+2iK_n\Psi_2\big], &
K^{\rm C,obs/emi}_{\Psi_2}&=\pm E_K\Psi_2,\nonumber\\
K^{\rm C,obs/emi}_{\delta_{T2}}&=\mp E_K\,\delta_{T2}, &
K^{\rm C,obs/emi}_{V_2}&=\pm E_K\,iK_nV_2,\nonumber\\
K^{\rm C,obs/emi}_{v^T_2+\omega_2}&=\pm E_K\,\nhat\cdot(\vv^T_2+\boldsymbol\omega_2)\quad(\text{helicity }\pm1), &
K^{\rm C,ray}_{\omega_2}&=E_K\,iK_n\,\nhat\cdot\boldsymbol\omega_2,\nonumber\\
K^{\rm C,ray}_{h_2}&=-\tfrac12E_K\,\dot h_{2,nn}\quad(\text{helicity }\pm2), &
K^{\rm C,ray}_{1\times1}&=4E_1E_2\big(T_{\Phi1}\dot T_{\Phi2}+T_{\Phi2}\dot T_{\Phi1}\big),
\label{eq:asm-chC}
\end{align}
upper sign at the observer, lower at emission.  The ray terms are $-\Gamma^0_{kk}$ evaluated on the second-order fields
($\tfrac12(\Phi_2'-\Psi_2')+\partial_n\Psi_2+\tfrac12\partial_n\omega_{2,n}-\tfrac14h'_{2,nn}$ in $L_2/2$; the $\partial_n\Psi_2$ term is
the second-order SW+ISW after integration by parts) and the local product $2\Phi_1\Phi_1'+2\Psi_1\Psi_1'+2(\Phi_1-\Psi_1)\partial_n\Psi_1$.
Grouping for the numerics (Table~\ref{tab:mono-groups}): $\Phi_2\Psi_2$ = ray $\Phi_2\Psi_2$ + endpoint $\Psi_2$; vector = endpoint
$v^T_2+\omega_2$ + ray $\omega_2$.

\paragraph{Type D (lensing, time delay, aberration+deflection, emission-surface shift, clock, observer displacement).}
\begin{align}
K^{\rm D}_{\rm lens}&=\int\!\dd\eta\sum_{(ab)}-P_\perp\mathcal D_{\perp,a}J_b+\Big[\sum_{(ab)}P_\perp\mathcal D_{\perp,a}U_b\Big]_{\eta_*}
&&(\Delta\vx_{1\perp}\cdot\nabla_\perp J,\ -\Delta\vx_{1\perp}\cdot\nabla_\perp U|_e),\nonumber\\
K^{\rm D}_{\rm delay}&=\int\!\dd\eta\sum_{(ab)}ik_b\mu_b\,\mathcal D_{r,a}J_b-\Big[\sum_{(ab)}ik_b\mu_b\,\mathcal D_{r,a}U_b\Big]_{\eta_*}
&&(\Delta x_{1r}\,\partial_nJ,\ -\Delta x_{1r}\,\partial_nU|_e),\nonumber\\
K^{\rm D}_{\rm defl}&=\int\!\dd\eta\sum_{(ab)}2P_\perp\mathcal F_a\TP(k_b)E_b-\Big[\sum_{(ab)}P_\perp\mathcal F_aT_V(k_b)E_b\Big]_{\eta_*}
&&(-2\vk_{1\perp}\cdot\nabla_\perp\Psi_1,\ \vk_{1\perp}\cdot\vv_{1\perp}|_e),\nonumber\\
K^{\rm D}_{\rm emi}&=-\Big[\sum_{(ab)}\delta\eta_e(\vk_a)\,\big(J_b+\mathbb D U_b\big)\Big]_{\eta_*},\qquad \mathbb D\equiv\partial_\eta-ik_b\mu_b
&&(\text{lower-endpoint shift}),\nonumber\\
K^{\rm D}_{\rm clock}&=\Big[\sum_{(ab)}\delta\eta_o(k_a)\,\big(J_b+\dot U_b\big)\Big]_{\eta_o}
&&(\text{upper-endpoint shift at fixed }\vx),\nonumber\\
K^{\rm D}_{\rm disp}&=-\Big[\sum_{(ab)}X_o(k_a)\,(\vk_a\cdot\vk_b)\,U_b\Big]_{\eta_o}
&&(\delta\vx_{o1}\cdot\nabla U|_o),
\label{eq:asm-chD}
\end{align}
with $\mathbb D=\partial_\eta-\nhat\cdot\nabla$ the derivative along the background ray; here, and in $\dot U_b$, $\partial_\eta$ is the
field derivative at fixed $\vx$, i.e.\ it acts on the transfer functions only, not on $E_b$ ($J_b,U_b$ evaluated at the time of the term).  Explicitly, the ray lensing kernel is $-2iP_\perp\sum_{(ab)}\mathcal D_{\perp,a}\TP(k_b)k_b\mu_bE_b$ and the ray time
delay $-2\sum_{(ab)}\mathcal D_{r,a}\TP(k_b)k_b^2\mu_b^2E_b$.  The radial shift $\nhat\,\delta\eta_{o1}$ and the observer displacement also
enter through $\mathcal D_{r}$, $\mathcal D_\perp$ \eqref{eq:asm-functionals}; the channels ``clock'' and ``disp'' contain only the
explicit endpoint pieces.

\begin{remark}[observer channels of the direct form]\label{rem:asm-obs}
At the observer $\chi=0$, so an obs channel is a polynomial in $\nhat$ of the degree of its $\mu_a$, $K_n$, $P_\perp$ content:
$K^{\rm C,obs}$, $K^{\rm D}_{\rm clock}$, $K^{\rm D}_{\rm disp}$ have degree $\le1$ and contribute to $\ell\le1$ only;
$K^{\rm B,obs}_{1\times1}$ has degree $2$ and contributes to $\ell\le2$.  For $\ell\ge2$ the observer motion and clock act only through
$\mathcal F,\mathcal D_\perp,\mathcal D_r$ and the type-B products.  (Consistent with the vanishing clock and displacement diagonals in
Table~\ref{tab:mono-groups}.)  The split into channels depends on the form of $L_2$ (in the shear form the explicit clock piece has degree 2,
Sec.~\ref{sec:theta2}); only the sum $K_2$ is form independent.
\end{remark}
The complete machine-generated set (fifteen channels, plus B) is App.~\ref{app:k2channels}.

\subsection{Squeezed limits: dilation and special conformal transformation}
\label{sec:asm-squeezed}
Let $\mathcal D\equiv\vk_2\cdot\partial_{\vk_2}$ (at fixed $\khat_2$, $\nhat$) and, for a vector $\vp$,
$\mathcal S_{\vp}\equiv(\vk_2\cdot\partial_{\vk_2})(\vp\cdot\partial_{\vk_2})-\tfrac12(\vp\cdot\vk_2)\nabla^2_{\vk_2}$.
\begin{theorem}[squeezed limit of the second-order map]\label{thm:asm-squeezed}
For $\eta_i\to0$ and every $\nhat$ (all multipoles, observer terms and normalisation included),
\begin{align}
K_2(\vk_1,\vk_2;\nhat)&=-\mathcal D K_1(\vk_2;\nhat)-\mathcal S_{\vk_1}K_1(\vk_2;\nhat)+O(k_1^2),\nonumber\\
\lim_{\vk_1\to0}K_2(\vk_1,\vk_2;\nhat)&=-\vk_2\cdot\partial_{\vk_2}K_1(\vk_2;\nhat).
\label{eq:asm-squeezed}
\end{align}
(For finite $\eta_i$ add $+\tfrac{\eta_i^2}6(\vk_1\cdot\vk_2)K_1(\vk_2;\nhat)$, a translation of the observer label.)  Since in RD
$K_1$ depends on $k$ only through $k\eta_o$, $k\eta_*$ and $\mu$, $\mathcal DK_1=(\eta_o\partial_{\eta_o}+\eta_*\partial_{\eta_*})K_1$.
\end{theorem}
\noindent\emph{Status.}  Derived from the adiabatic-mode transformation of the full observable, independently of
$K_2$: the long mode $\zeta_L=c+\vq\cdot\vx$ is the Poisson-gauge image of FRW under
$\tilde\eta=\eta-\tfrac\eta3\zeta_L$, $\tilde\vx=(1+\zeta_L)\vx-\tfrac{x^2}2\vq-\tfrac{\eta^2}6\vq$; every ingredient of $T_{\rm obs}=T_oe^{-S}$
(shear, null geodesic, $T_{\rm phys}=T_*$ surface, tagged worldline, proper-time clock, sky normalisation) is covariant, and at
$\eta\to0$ the map is conformal with $\tilde\zeta_S(\tilde\vx)=\zeta_S(\vx(\tilde\vx))$ exactly.  Verified on the full $K_2$
\eqref{eq:asm-K2} (SQ-2): $\|K_2(s\hat\ve,\vk_2)+\mathcal DK_1\|$ (rms over $\ell\ge2$) $=3.18\times10^{-1,-2,-3}$ for
$s=10^{-2,-3,-4}$ ($|\vk_2|=1.07$; the dilation itself has rms $11.25$), exactly linear in $s$; the $k_1$-odd part
$[K_2(s\hat\ve)-K_2(-s\hat\ve)]/2s$ equals $-\mathcal S_{\hat\ve}K_1$ to $1.5$--$6\times10^{-5}$ ($|\vk_2|=1.07$) and $1.6$--$2.2\times10^{-4}$
($|\vk_2|=2.44$) per multipole, the accuracy of the finite-difference Hessian.  Individual channels are $O(10)$ at $s\to0$
(e.g.\ ray $V_2$-containing terms $11.0$, time delay $13.9$) and cancel to the dilation.

\paragraph{Why a constant $\zeta$ is a dilation and not zero.}  A constant $\zeta_{\rm nl}=c$ is pure gauge: the early-time
spatial metric $a^2e^{2\zeta}\dd\vx^2$ is FRW in the coordinates $\tilde\vx=e^c\vx$, and the observer (at the origin), its proper-time
clock and the emission temperature $T_*$ are invariant.  But $\zeta_{\rm nl}=c+\zeta_S$ is then the configuration
$\tilde\zeta_S(\tilde\vx)=\zeta_S(e^{-c}\tilde\vx)$ in FRW coordinates, i.e.\ the \emph{same} observable evaluated on the short field
\emph{relabelled} to $e^{c}$ times longer comoving wavelengths: $\Theta[c+\zeta_S]=\Theta[\zeta_S(e^{-c}\,\cdot)]$, whose $O(c)$ term is
$-c\,\vk\cdot\partial_\vk K_1$.  Physically, the long mode stretches every short wavelength relative to the only other scales in
the problem, the emission and observation times fixed by $T_{\rm phys}=T_*$ and by the proper-time clock; since the RD map depends
on $k\eta_*$, $k\eta_o$ only, this is the same as $-c(\eta_o\partial_{\eta_o}+\eta_*\partial_{\eta_*})$.  For a \emph{fixed} Gaussian
realisation of $\zeta_S$ the relabelled field is a different realisation, so $K_2(0,\vk_2)$ cannot vanish; ``unobservable'' means only
that the long mode is equivalent to a relabelling (Maldacena's consistency relation for the map), whose effect on the statistics
vanishes iff $k^3P$ is scale invariant.  The separate CPV-type pieces (amplitude, emission-time shift $\eta_e=\eta_*(1+c/3)$, clock
$\eta_o(1+c/3)$, coordinate length of the ray, field cross terms, monopole normalisation) do not survive individually.  At $O(k_1)$ the
observer velocity, drift, emission-surface tilt, gradient lensing/time delay and the squeezed vector mode cancel likewise, leaving the
SCT relabelling $\mathcal S_{\vk_1}$; physical effects start at $O(k_1^2)$.

\paragraph{Consequence for the IR of the loop (second-order half).}  Let $q_*$ separate soft from hard momenta and
$\sigma_L^2\equiv\int_{k<q_*}\frac{\dd^3k}{(2\pi)^3}P(k)$.  In \eqref{eq:loops} the regions $k_1<q_*$ and $k_2<q_*$ contribute equally;
with \eqref{eq:asm-squeezed} ($O(k_1)$ terms vanish by parity after the angular integral) and \eqref{eq:asm-K1lm},
\begin{equation}
\tfrac14C^{(22)}_\ell\big|_{\rm soft}=\sigma_L^2\,\frac1{2\ell+1}\sum_m\int\!\frac{\dd^3k}{(2\pi)^3}P(k)\,\big|\mathcal DK_{1,\ell m}(\vk)\big|^2
=\sigma_L^2\,\frac2\pi\int k^2\dd k\,P(k)\,\big[k\partial_k\Delta_\ell(k)\big]^2+O(q_*^2\sigma_L^2),
\label{eq:asm-IR22}
\end{equation}
the second-order dilation response.  It is positive, so for $P\propto k^{-3}$ ($\sigma_L^2\propto\int\dd q/q$) $C^{(22)}_\ell$ alone is
logarithmically IR divergent and nothing inside $C^{(22)}$ can cancel it.  The cancellation requires the $(13)$ soft region with
$K_3(\vq,-\vq,\vk)\to\mathcal D^2K_1(\vk)$, derived from the same adiabatic-mode transformation and verified on the full contracted kernel
(Theorem~\ref{thm:asm-squeezed3}),
after which the sum is $\sigma_L^2\cdot\tfrac12\cdot\tfrac2\pi\int k^2\dd k\,\Delta_\ell^2(3+\mathcal D)^2P$: a total derivative in $\ln k$ that
vanishes for scale-invariant $k^3P$, and leaves an IR sensitivity $\propto\sigma_L^2$ times the running of $k^3P$ otherwise
(Sec.~\ref{sec:asm-IR}, \eqref{eq:asm-IRtot}).  For the monochromatic spectrum of Sec.~\ref{sec:mono} there is no soft region; the corner $K=|\vk_1+\vk_2|\to0$
treated there is a different limit.

\subsection{Contracted third-order kernel}
\label{sec:asm-K3c}
\paragraph{Definition and factors.}  With $\Theta^{(3)}(\nhat)=\int\!\!\int\!\!\int\zeta\zeta\zeta\,K_3$ \eqref{eq:Kr} ($K_3$ symmetric), the only
part of $K_3$ that enters $C^{(13)}_\ell$ is its contraction with one power spectrum.  We define the \emph{contracted third-order kernel}
as the mean response of $\Theta^{(3)}$ to one Fourier mode,
\begin{equation}
\kappa(\vk;\nhat)\equiv\Big\langle\frac{\delta\Theta^{(3)}(\nhat)}{\delta\zeta_\vk}\Big\rangle
=3\int\!\frac{\dd^3q}{(2\pi)^3}P(q)\,K_3(\vk,\vq,-\vq;\nhat)\equiv3K_{3c}(\vk;\nhat),
\label{eq:kappa-def}
\end{equation}
($\delta\zeta_{\vk'}/\delta\zeta_\vk=(2\pi)^3\delta^3(\vk-\vk')$); the factor 3 counts the three ways of choosing the leg that carries $\vk$, so
$\kappa$ is the contracted map $\Theta^{(3)}_c$ of \eqref{eq:Theta3c}, $K_{3c}=\kappa/3$ is the single-pairing contraction, and the transfer
function $\Delta^{(3c)}_\ell$ of \eqref{eq:Delta3c} (which includes the factor 3) is the Legendre coefficient of $\kappa$:
\begin{equation}
\kappa(\vk;\nhat)=\sum_\ell(2\ell+1)\,i^\ell\,\Delta^{\kappa}_\ell(k)\,P_\ell(\nhat\cdot\khat),\qquad \Delta^\kappa_\ell\equiv\Delta^{(3c)}_\ell,\qquad
C^{(13)}_\ell=\frac2\pi\int\!k^2\dd k\,P(k)\,\Delta_\ell(k)\,\Delta^{\kappa}_\ell(k)^* .
\label{eq:asm-C13}
\end{equation}
In practice $\kappa$ is computed by polarisation: for $\zeta=a\,e^{i\vk\cdot\vx}+b\,e^{i\vq\cdot\vx}+c\,e^{-i\vq\cdot\vx}$ with $a^2=b^2=c^2=0$ the
coefficient of $abc$ in $\Theta^{(3)}$ is ${\rm coef}_{abc}(\vk,\vq,-\vq;\nhat)=3!\,K_3(\vk,\vq,-\vq;\nhat)$, hence
\begin{equation}
\kappa(\vk;\nhat)=\frac12\int\!\frac{\dd^3q}{(2\pi)^3}P(q)\,{\rm coef}_{abc}(\vk,\vq,-\vq;\nhat).
\label{eq:asm-kappa-coef}
\end{equation}
$\Theta^{(3)}$ is \eqref{eq:theta3}, $m_3=L_3+3L_1L_2+L_1^3$, with $L_3$ in the shear form (App.~\ref{app:L3full}; the groups are listed with the
contraction rule in App.~\ref{app:k3channels}).  After the $\qhat$ integration $\kappa$ depends on $\nhat$ only through $\nu=\nhat\cdot\khat$.

\paragraph{Second-order observer objects.}  The third-order map needs the clock and the worldline at second order.  In the limit
$\eta_i\to0$ of \eqref{eq:dxo}, \eqref{eq:deta_o2} (label at the initial event; all fields on the worldline $\vx=\mathbf 0$),
\begin{align}
\tfrac12\delta\eta_{o2}&=-\frac1{\eta_o}\Big[\tfrac12\delta\eta_{o1}^2+\eta_o\,\delta\eta_{o1}\Psi_1\big|_o+\int_0^{\eta_o}\!\eta\,\Big(\tfrac12\Psi_2-\tfrac12\Psi_1^2-\tfrac12|\vv_1|^2
+\mathbf X_1\cdot\nabla\Psi_1\Big)\dd\eta\Big],\nonumber\\
\tfrac12\delta\vx_{o2}&=\int_0^{\eta_o}\!\Big[\tfrac12\vv_2+(\mathbf X_1\cdot\nabla)\vv_1\Big]\dd\eta+\delta\eta_{o1}\,\vv_1\big|_o,\qquad \mathbf X_1(\eta)\equiv\int_0^\eta\vv_1\,\dd\eta',
\label{eq:asm-clock3}
\end{align}
with $\delta\eta_{o1}$, $\delta\vx_{o1}=\mathbf X_1(\eta_o)$ of \eqref{eq:asm-clock}, \eqref{eq:asm-disp}.  (Check: for the dilation family, $\lambda=1+\eps a_1+\tfrac{\eps^2}2a_2$,
$a_1=-c/3$, the first line gives $\tfrac12\delta\eta_{o2}=\eta_o(c^2/9-a_2/2)$, the second-order coefficient of $\bar\eta_o/\lambda$.)  The fields in
\eqref{eq:asm-clock3} include the tadpoles: in the configuration $(\vk,\vq,-\vq)$ the worldline integral of $\Psi_2$ contains the pair
$(\vq,-\vq)$ multiplying $K_1(\vk)$, and the pairs $(\vk,\pm\vq)$ multiplying first-order objects of $\mp\vq$.  The second-order ray
functionals $\vk_2$, $\Delta\vx_2$ are \eqref{eq:k2def}, \eqref{eq:Dx2def} with the sign of Remark~\ref{rem:signslip}, and the emission shift
$\delta\eta_{e2}$ is \eqref{eq:deta_e}.

\paragraph{Channel classes.}  Let $\langle Y Z\rangle\equiv\int\frac{\dd^3q}{(2\pi)^3}P(q)\,Y[\vq]\,Z[-\vq]$ denote the contraction of two
first-order objects (fields or ray functionals, possibly at different points of the ray), $X[\vk]$ the kernel of a first-order object
($\zeta_\vk\,T_X(k,\eta)E$ with the Fourier rules of Sec.~\ref{sec:asm-K2}), $X^{\rm s}_2(\vq_1,\vq_2)\equiv K_X(\vq_1,\vq_2)+K_X(\vq_2,\vq_1)$ the
pair kernel of a second-order object and $\langle X_2\rangle=\int\frac{\dd^3q}{(2\pi)^3}P(q)K_X(\vq,-\vq)$ its tadpole \eqref{eq:tad}.  Applying
\eqref{eq:asm-kappa-coef} monomial by monomial gives three classes.
\begin{itemize}
\item[(i)] \emph{Cubic first-order products} $ABC$ (one self-contraction): the leg $\vk$ sits on one factor and the other two are contracted,
\begin{equation}
\kappa^{\rm(i)}[ABC]=A[\vk]\,\langle BC\rangle+B[\vk]\,\langle AC\rangle+C[\vk]\,\langle AB\rangle ,
\label{eq:asm-classi}
\end{equation}
a variance (equal-point for local products, an unequal-point correlation along the ray for products with ray functionals) times a first-order
map.  Example: the cube $L_1^3$ of $m_3$ gives $3K_1(\vk;\nhat)\langle L_1L_1\rangle(\nhat)$, and $\langle L_1L_1\rangle=\langle\sky{m_1}^2\rangle+\sum_{\ell\ge1}\frac{2\ell+1}{4\pi}C^{(11)}_\ell$
is direction independent, $=\xi^{(11)}(0)$ \eqref{eq:xi0}.
\item[(ii)] \emph{First $\times$ second order} $A_1B_2$ (including first-order fields times second-order ray functionals, clock, displacement
and emission shift):
\begin{equation}
\kappa^{\rm(ii)}[A_1B_2]=\underbrace{A[\vk]\,\langle B_2\rangle}_{\rm tadpole\ pair}
+\underbrace{\int\!\frac{\dd^3q}{(2\pi)^3}P(q)\,A[\vq]\,B^{\rm s}_2(\vk,-\vq)}_{\rm cross\text{-}contracted}.
\label{eq:asm-classii}
\end{equation}
The tadpole term multiplies the free leg only, so only the $\qhat$-averaged tadpole enters $\kappa$ (Sec.~\ref{sec:asm-C13}).  The cross term contains
the second-order kernels at the momenta $(\vk,-\vq)$, $|\vk-\vq|=p$, with all their Poisson inversions $p^{-2}$, $p^{-4}$.
\item[(iii)] \emph{Third-order fields}, $X_3\to K_{3c,X}(\vq)+K_{3c,X}(-\vq)$, i.e.\ the contracted solution $X_{3c}$ of \eqref{eq:contracted},
\eqref{eq:third_eqs}: $\Phi_{3c}=\int\frac{\dd^3q}{(2\pi)^3}P(q)\big[\int_0^\eta G_\Phi K_{3c,S_\Phi}\dd\eta'+C_3\,U(k\eta/\sqrt3)\big]$, $\Psi_{3c}=\Phi_{3c}-Q_{3c}$,
$V_{3c}=-\tfrac{\eta^2}4[2(\Phi'_{3c}+\Psi_{3c}/\eta)+K_{3c,S_{0z}}/(ik)]$.  In the shear form only the velocity enters (the contracted vector and
tensor parts vanish by isotropy), $\vv_{3c}=i\vk V_{3c}$, through the same projection as at first order:
\begin{equation}
\kappa^{\rm(iii)}(\vk;\nhat)=\int_{\eta_*}^{\eta_o}\!\dd\eta\,\tfrac23k^2\,P_2(\nhat\cdot\khat)\,V_{3c}(k,\eta)\,e^{i\vk\cdot\nhat\chi},\qquad
\Delta^{\rm(iii)}_\ell=\int\!\dd\eta\,\tfrac23k^2V_{3c}\sum_{J}(2J+1)\threej{\ell}{2}{J}{0}{0}{0}^2i^{J-\ell}j_J(k\chi),
\label{eq:asm-classiii}
\end{equation}
by (A-$L$) \eqref{eq:AL} with $L=2$.  (In the direct form, $\Phi_{3c},\Psi_{3c},\delta_{T3c},V_{3c}$ appear instead; the two forms agree, below.)
\end{itemize}
The normalisation terms of \eqref{eq:theta3} contribute, for $\ell\ge1$ (with $K^{m_2}\equiv2\,{\rm Sym}[K_{L_2/2}]+K_1K_1$ the kernel of $m_2$ and
$\bar K_2$ its sky average, Sec.~\ref{sec:thirteen}),
\begin{equation}
\kappa_{\rm norm}=-6\!\int_{\vq}\!P\,\Delta_0(q)\,K^{m_2}(\vk,-\vq;\nhat)-3\langle\!\langle m_2\rangle\!\rangle K_1(\vk;\nhat)
-6\!\int_{\vq}\!P\,\bar K_2(\vk,-\vq)\,K_1(\vq;\nhat)+6\langle\sky{m_1}^2\rangle K_1(\vk;\nhat),
\label{eq:asm-kappanorm}
\end{equation}
$\int_\vq\equiv\int\frac{\dd^3q}{(2\pi)^3}$; inserted in \eqref{eq:asm-C13} they give the four terms of (E0) \eqref{eq:E0} with (E2), (E3)
\eqref{eq:E2E3}.  The complete list of groups ($L_1$: 3 terms; $\tfrac12L_2$: 34; $\tfrac16L_3$: 359 in 32 groups) with their classes is
App.~\ref{app:k3channels}; the engine evaluates every group in a per-direction frame $(\ehat_1,\ehat_2,\nhat)$, builds the ray functionals by
spectral cumulative integration on a Chebyshev grid and the third-order fields from Green's-function quadrature.

\paragraph{The spatial triad.}  The direction map \eqref{eq:dirmap} contains, from second order on, the spatial triad of the observer
\eqref{eq:triad} through $k_{2,\rm init}$ \eqref{eq:kinit2}.
\begin{lemma}[triad independence]\label{lem:asm-triad}
(a) $\Theta^{(1)}$ and $\Theta^{(2)}$ do not depend on the choice of orthonormal triad at the observer among those that agree at first order.
(b) Let two such triads differ by a rotation $\mathsf R=1+\tfrac{\eps^2}2\Omega+O(\eps^3)$, $\Omega_{ij}=\epsilon_{ijk}\omega_k$, with $\boldsymbol\omega$
independent of $\nhat$ (a \emph{rigid} rotation; $\boldsymbol\omega$ may depend on the fields at the observer).  Then
$\Theta^{(3)}\to\Theta^{(3)}+3(\Omega\nhat)\cdot\nabla_{\nhat}\Theta^{(1)}=\Theta^{(3)}+3i\,(\boldsymbol\omega\cdot\mathbf L)\Theta^{(1)}$, $\mathbf L=-i\nhat\times\nabla_{\nhat}$
(up to the sign convention of $\Omega$), and $\Re C^{(13)}_\ell$ is unchanged.
\end{lemma}
\emph{Proof.}  The triad enters only through the map from the observed direction to the initial photon direction, $\nhat\to\mathsf R\nhat$.
Since every triad of the Gram--Schmidt type agrees at first order \eqref{eq:triad}, the change is $O(\eps^2)$ in the direction, which changes
$\Theta=\eps\Theta^{(1)}+\dots$ first at $O(\eps^3)$: $\Theta(\mathsf R\nhat)-\Theta(\nhat)=\tfrac{\eps^3}2(\Omega\nhat)\cdot\nabla_{\nhat}\Theta^{(1)}+O(\eps^4)$,
i.e.\ $\delta\Theta^{(3)}=3(\Omega\nhat)\cdot\nabla_{\nhat}\Theta^{(1)}$, and $(\Omega\nhat)\cdot\nabla_{\nhat}=\boldsymbol\omega\cdot(\nhat\times\nabla_{\nhat})$.
For each realisation $\sum_ma^{(1)*}_{\ell m}[(\boldsymbol\omega\cdot\mathbf L)a^{(1)}]_{\ell m}$ is real ($\mathbf L$ is Hermitian and acts within each $\ell$,
$\boldsymbol\omega$ real), so the change of $\sum_ma^{(1)*}_{\ell m}a^{(3)}_{\ell m}$ is purely imaginary and $\Re C^{(13)}_\ell$ is invariant.  $\square$

\noindent The triad fixed by Gram--Schmidt orthonormalisation of the \emph{global} coordinate basis (the canonical triad of
Sec.~\ref{sec:ray}) is one triad for the whole sky, and any other global choice differs from it by a rigid rotation.  A triad constructed per
direction (Gram--Schmidt in the frame adapted to each $\nhat$, e.g.\ the $\nhat=\hat{\mathbf z}$ split of the term lists) differs from it by an
$\nhat$-dependent rotation, which is not a rotation of the sky map: it changes $\kappa$ (at the $0.3\%$ level in the monochromatic case) and
must not be used at third order.  We use the canonical global triad, evaluated in global components for each $\nhat$ and then rotated into the
per-direction frame.  (In the shear form the observer contains no $k_2$ term, so the canonical triad enters only through the ray functional
$\vk_2$; the direct-form observer terms written in the $\nhat=\hat{\mathbf z}$ split implicitly use the per-direction triad and must be
re-expressed before they are compared.)

\subsection{\texorpdfstring{$C_\ell^{(13)}$}{C13}}
\label{sec:asm-C13}
Combining \eqref{eq:asm-C13} with the classes,
\begin{equation}
C^{(13)}_\ell=\frac2\pi\int\!k^2\dd k\,P(k)\,\Delta_\ell(k)\Big[\Delta^{\rm(i)}_\ell+\Delta^{\rm(ii)}_\ell+\Delta^{\rm(iii)}_\ell+\Delta^{\rm norm}_\ell\Big]^*(k),
\label{eq:asm-C13split}
\end{equation}
where each $\Delta^{(\cdot)}_\ell$ is the Legendre coefficient of the corresponding part of $\kappa$; equivalently, in the language of
Sec.~\ref{sec:thirteen},
$C^{(13)}_\ell=T_{3,\ell}-3T_{\sky12,\ell}-3T_{\sky21,\ell}+6\langle\sky{m_1}^2\rangle C^{(11)}_\ell$ (E0), with $T_3$ the $m_3$ part
($T^{\rm(i)}$, $T^{\rm(ii)}$, $T^{\rm(iii)}$) and the normalisation pieces from \eqref{eq:asm-kappanorm}.  The radial functions are named
objects: $\Delta^{\rm(iii)}_\ell$ in \eqref{eq:asm-classiii} (one declared time integral inside $V_{3c}$); $\Delta^{\rm(i)}_\ell$ from the
variances $\xi^{XY}(0)$ \eqref{eq:xi0} and the unequal-point correlations of the ray functionals (reduction (E1) with $\Delta\chi\ne0$);
$\Delta^{\rm(ii)}_\ell$ from the tadpoles $\langle X_2\rangle(\eta)$ (closed forms, App.~\ref{app:kernels}) and from the second-order pair kernels
at $(\vk,-\vq)$, reduced by (E1) and (A-$L$).  None of the $\vq$ integrals has been done in closed form; for the monochromatic spectrum they
collapse to one angular integral (Sec.~\ref{sec:mono-13}).

\paragraph{Validation of $\kappa$} (App.~\ref{app:ledger}, R-9--R-11, EF-22, H-13).
\begin{enumerate}\itemsep1pt
\item \emph{Shear form $=$ direct form at third order.}  The two term lists of $\tfrac16L_3$, evaluated with the contracted third-order fields
\eqref{eq:third_eqs}, the second-order functionals, clock, displacement and emission shift, give the same $\qhat$-azimuth-averaged
${\rm coef}_{abc}(\vk,\vq,-\vq;\nhat)$ to $2\times10^{-11}$ absolute ($3\times10^{-14}$ relative) at $\mu=\khat\cdot\qhat=0.31$ and $-0.62$.  Since the
direct form contains $\Phi_{3c},\Psi_{3c},\delta_{T3c},V_{3c}$ and the shear form only $V_{3c}$, this tests the contracted third-order solution
(third-order conservation), both term lists and all third-order ray objects at once.  At lower order: $K_1$ from the $L_1$ lists equals the
type-A expansion with $\Delta_\ell$ to $2\times10^{-15}$; $\tfrac12L_2$ shear $=$ direct for three generic momenta, all pairs including the $K=0$
pair, to $10^{-13}$; the second-order null condition holds for the computed $\vk_2$ to $10^{-13}$.
\item \emph{Residual-gauge invariance.}  At $K=0$ Poisson gauge leaves a time reparametrisation $\eta\to\eta+\tfrac{\eps^2}2\tau(\eta)$ free; it
shifts the isotropic tadpoles, $\delta\langle\Psi_2\rangle=\tau'+\tau/\eta$, $\delta\langle\Phi_2\rangle=-\tau/\eta$, $\delta\langle\delta_2\rangle=-4\tau/\eta$
($\zeta_{\rm nl}$ unchanged), and the third-order fields by $\delta\Phi_3=3\tau T_\Phi'+6(\tau/\eta)T_\Phi$, $\delta\Psi_3=3\tau T_\Phi'+6(\tau'+\tau/\eta)T_\Phi$,
$\delta V_3=3\tau T_V'+3\tau'T_V$.  (a) The contracted sources with shifted tadpoles, solved with the same Green's function and the shifted
amplitude $\delta C_3=\delta\Phi_3(0)$, reproduce these shifts to $10^{-13}$: the tadpole bookkeeping of the third-order solution is gauge
covariant.  (b) The fully assembled $\kappa$ (all tadpole routes: local $2{+}1$ products, radial $\vk_2$ and $\Delta\vx_2$, clock $\delta\eta_{o2}$,
emission shift $\delta\eta_{e2}$, class (iii), $3L_1L_2$ and $-3\sky{m_2}m_1$) is unchanged to $6\times10^{-14}$ for $\tau=0.1\eta$ and $3\times10^{-13}$
for $\tau=0.1\eta^3$ (scale of $\kappa$: $4\times10^2$), while individual channels change by $O(1$--$4)$ (time delay with $\Delta\vx_2$: $3.6$; clock:
$2.3$; class (iii): $1.35$; local $2{+}1$ and radial $\vk_2$: $1.18$ each; emission shift: $0.15$).
\item \emph{Reality.}  $\Im C^{(13)}_\ell\sim10^{-14}$ in the monochromatic evaluation (relative to $\Re C^{(13)}_\ell$, $\ell\le12$).
\end{enumerate}

\paragraph{The tadpole (mean-field) convention.}  Item 2 settles the zero-mode question raised by \eqref{eq:KPhi20_smallk} in the gauge sense.
(1) In \eqref{eq:asm-classii} the tadpole multiplies first-order objects of the free leg $\vk$ only, so only the $\qhat$-average of the $K=0$
pair kernel can enter $\kappa$: the anisotropic (quadrupolar, $q^2(3\mu^2-1)/k^2$) part of \eqref{eq:KPhi20_smallk}, a constant anisotropic shear,
drops out identically, without any convention.  (2) The isotropic tadpole has three functions $(\langle\Phi_2\rangle,\langle\Psi_2\rangle,\langle\delta_2\rangle)(\eta)$
subject to two $K=0$ constraints; the remaining freedom is exactly $\tau(\eta)$ (plus the constant dilation fixed by $\zeta^{(2)}(0)=0$), under
which $\kappa$ is invariant.  (3) The sourced part of the tadpole is not a gauge mode: its source is the $\qhat$-averaged $K=0$ stress of the
short modes, and it is the physical back-reaction of the short modes on the homogeneous expansion; it is kept and is not expected to cancel.
In the shear form, $L_2$ contains no tadpole field (the shear of a homogeneous perturbation vanishes), so $3L_1L_2$ and $-3\sky{m_2}m_1$ see the
pair $(\vq,-\vq)$ only through products of first-order objects (class (i)); the tadpole fields act only through the third-order routes listed
in item 2.
\pending{an analytic proof of zero-mode invariance for general momenta (all anisotropic $K\to0$ pieces, not only the $\qhat$-averaged ones);
the numerical statements above cover the residual time reparametrisation and the plane-wave configurations of Sec.~\ref{sec:check_dilation}.}

\subsection{Infrared and ultraviolet structure of the loop}
\label{sec:asm-IR}
Write $\kappa(\vk;\nhat)=\int\frac{q^2\dd q}{2\pi^2}P(q)\,\tilde\kappa(\vk,q;\nhat)$, $\tilde\kappa\equiv3\langle K_3(\vk,\vq,-\vq;\nhat)\rangle_{\qhat}$, and
$\Delta^{\tilde\kappa}_\ell(k,q)$ for its Legendre coefficients.
\begin{theorem}[squeezed limit at third order]\label{thm:asm-squeezed3}
For $\eta_i\to0$, with $\mathcal D=\vk\cdot\partial_\vk$,
\begin{equation}
K_3(\vq,-\vq,\vk;\nhat)=\mathcal D^2K_1(\vk;\nhat)+O(q^2),\qquad\text{hence}\qquad \Delta^{\tilde\kappa}_\ell(k,q)=3\,\mathcal D^2\Delta_\ell(k)+O(q^2).
\label{eq:asm-squeezed3}
\end{equation}
\end{theorem}
\noindent\emph{Derivation.}  As for Theorem~\ref{thm:asm-squeezed}, two long modes $\vk_a,\vk_b$ compose to the adiabatic transformation with
$c=\zeta_L$ to second order; the composite of the dilation $e^{-c\mathcal D}$ and the special conformal transformation gives, to all orders in
$c$, $K_{m+1}(0,\dots,0,\vk)=(-\mathcal D)^mK_1(\vk)$, and with two long legs
$K_3(\vk_a,\vk_b,\vk)=\mathcal D^2K_1+(\mathcal S_{\vk_a}+\mathcal S_{\vk_b})\mathcal DK_1+O(k_a^2,k_b^2,k_ak_b)$ (plus an $\eta_i^2$ label term).  For
$\vk_b=-\vk_a=-\vq$ the $O(q)$ terms cancel.  Physically: a constant $\zeta=c$ is the same universe in the coordinates $\vx\to e^c\vx$, so
$\Theta[c+\zeta_S]=\Theta[\zeta_S(e^{-c}\,\cdot)]$, whose $O(c^2)$ term is $\tfrac12c^2\mathcal D^2$ of the first-order map.
\emph{Numerical verification on the full assembled $\kappa$} (including normalisation; $k=1$, $\eta_*=2$, $\eta_o=8$; SQ-3):
\begin{center}\footnotesize
\begin{tabular}{r|rrrr}
\hline
& $\ell=1$ & $\ell=2$ & $\ell=3$ & $\ell=4$\\ \hline
$q=0.100$ & $25.523$ & $-15.785$ & $-10.342$ & $-0.218$\\
$q=0.050$ & $25.685$ & $-16.210$ & $-10.255$ & $0.197$\\
$q=0.025$ & $25.732$ & $-16.322$ & $-10.240$ & $0.301$\\ \hline
$3\mathcal D^2\Delta_\ell$ & $25.748$ & $-16.360$ & $-10.235$ & $0.335$\\ \hline
\end{tabular}
\end{center}
The deviations decrease by a factor $4.0$ per halving of $q$ ($\ell=2$: $0.575$, $0.151$, $0.038$; $\ell=4$: $0.553$, $0.138$, $0.034$), i.e.\ as $q^2$.

\paragraph{Combined infrared behaviour.}  Let $q_*$ separate soft from hard momenta, $\sigma_L^2=\int_{q<q_*}\frac{\dd^3q}{(2\pi)^3}P(q)$.  By
\eqref{eq:asm-squeezed3} the soft region of $\tfrac13\Re C^{(13)}_\ell$ is $\sigma_L^2\frac2\pi\int k^2\dd k\,P\,\Delta_\ell\,\mathcal D^2\Delta_\ell$; adding the
soft region \eqref{eq:asm-IR22} of $\tfrac14C^{(22)}_\ell$ and using $(\mathcal D\Delta)^2+\Delta\mathcal D^2\Delta=\tfrac12\mathcal D^2(\Delta^2)$ and two
integrations by parts in $\ln k$,
\begin{equation}
\Big[\tfrac14C^{(22)}_\ell+\tfrac13\Re C^{(13)}_\ell\Big]_{\rm soft}
=\sigma_L^2\,\frac2\pi\int\!\dd\ln k\;\tfrac12\,\Delta_\ell(k)^2\,\frac{\dd^2}{\dd(\ln k)^2}\big(k^3P(k)\big)+O(q_*^2\sigma_L^2),
\label{eq:asm-IRtot}
\end{equation}
provided the boundary terms $\Delta_\ell^2\,k^3P$, $\Delta_\ell\Delta_\ell'k^4P$ vanish at $k\to0$ and $k\to\infty$ (true for $\ell\ge2$ and the power laws
below: $\Delta_\ell=O(k^\ell)$ at small $k$ and $O(k^{-1})$ at large $k$).  Each half separately is $\propto\sigma_L^2$ with a coefficient of
definite sign; the sum is a total second derivative of $k^3P$.  For a power law $P\propto k^{n_s-4}$, $\dd^2_{\ln k}(k^3P)=(n_s-1)^2k^3P$ and
\begin{equation}
\Big[\tfrac14C^{(22)}_\ell+\tfrac13\Re C^{(13)}_\ell\Big]_{\rm soft}=\tfrac12(n_s-1)^2\,\sigma_L^2\,C^{(11)}_\ell+O(q_*^2\sigma_L^2),\qquad
\sigma_L^2\propto\int_0^{q_*}\!\dd q\,q^{n_s-2}.
\label{eq:asm-IRns}
\end{equation}
Hence: for $n_s=1$ the coefficient of the (logarithmically divergent) $\sigma_L^2$ vanishes identically and the loop is IR finite; for
$n_s>1$, $\sigma_L^2$ converges and the loop is IR finite; for $n_s<1$, $\sigma_L^2$ diverges as $q_{\min}^{n_s-1}$ and the one-loop spectrum has
a genuine IR sensitivity $\tfrac12(n_s-1)^2\sigma_L^2C^{(11)}_\ell$ (the response to a long-wavelength dilation of a spectrum that is not scale
invariant).  The $O(q^2)$ remainder converges at $q\to0$ for $n_s>-1$.  The monochromatic spectrum has no soft region.  The formula
\eqref{eq:asm-IRtot} is derived; it has not been tested numerically for a tilted spectrum.

$\Delta^{\tilde\kappa}_\ell(k,q)$ is

2 & $-16.36$ & $-4.68$ & $2.04$ & $5.96$ & $11.00$\\

($q\to0$: the squeezed value $3\mathcal D^2\Delta_\ell$; the $\qhat$ quadrature needs $\gtrsim2q\chi_*+8$ azimuthal nodes).  It stays $O(10)$ and

Power counting supports $O(q^0)$: inside the horizon the potentials decay, $T_\Phi\sim(q\eta)^{-2}$, but $\delta_T$ and $v$ stay $O(1)$ (no

$\tilde\kappa=O(q^0)$ as $q\to\infty$, the $q$ integral $\int\dd q\,q^2P(q)\tilde\kappa$ converges in the UV for $n_s<1$ and is logarithmically

No analytic large-$q$ expansion of $K_{3c}$ has been made; the UV statement is unresolved (Sec.~\ref{sec:unresolved}).
