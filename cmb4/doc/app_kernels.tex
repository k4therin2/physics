\section{Explicit kernels (machine generated)}
\label{app:kernels}
All expressions below are exported directly from the computer-algebra solution of the field equations (App.~\ref{app:repro}).
Notation: $T_q\equiv\TP(q\eta/\sqrt3)$, $\dot T_q\equiv\partial_\eta T_q$,
$p=|\vk-\vq|$, $\mu=\khat\cdot\qhat=(k^2+q^2-p^2)/(2kq)$; every second-order kernel is per $\zeta_{\vq}\zeta_{\vp}$ in the sense of
\eqref{eq:kernel_def}.  Sources are those of $E^{(2)}_{\mu\nu}=2[\eps^2]E_{\mu\nu}$ in the frame $\vk=k\hat{\mathbf z}$,
$\vq$ in the $x$--$z$ plane.  The transfer functions \eqref{eq:transfer} and the dictionary $\dot T_q=\partial_\eta\TP$ with
$\partial_\eta=(q/\sqrt3)\,\dd/\dd x$ convert everything to closed form in $\sin,\cos$ of $q\eta/\sqrt3$, $p\eta/\sqrt3$.

\paragraph{$K_{S_\Phi}$ (source of $\mathcal L_\Phi\Phi_2=S_\Phi$)}

\paragraph{$G_\Phi(k;\eta,\eta')$}
\begin{equation*}
G_\Phi(k;\eta,\eta')=\frac{\sqrt3}{k}\,\frac{x'}{x^3}\Big[(1+xx')\sin(x-x')-(x-x')\cos(x-x')\Big],\qquad x=\frac{k\eta}{\sqrt3},\ x'=\frac{k\eta'}{\sqrt3}
\end{equation*}

\begin{dmath}
\frac{\eta' \sin{\left(k \left(- \eta' + \eta\right) \right)}}{\eta k}
\end{dmath}

\begin{dmath}
\frac{{\eta'}^{2}}{\eta^{2}}
\end{dmath}
