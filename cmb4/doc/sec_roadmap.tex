\section{Roadmap: target, structure, status}
\label{sec:roadmap}

\noindent\fbox{\begin{minipage}{0.97\textwidth}\small
\textbf{Target and conventions.}
$C_\ell=\eps^2C^{(11)}_\ell+\eps^4[\tfrac14C^{(22)}_\ell+\tfrac13\Re C^{(13)}_\ell]+O(\eps^6)$, $\ell\ge2$ \eqref{eq:cl_target}, with
$C^{(rs)}_\ell=\frac1{2\ell+1}\sum_m\conn{a^{(r)}_{\ell m}a^{(s)*}_{\ell m}}$, $\conn{\cdot}$ connected in the $a_{\ell m}$ \eqref{eq:conn}.
Fourier $f=\int\frac{\dd^3k}{(2\pi)^3}e^{i\vk\cdot\vx}f_\vk$, $\langle\zeta_\vk\zeta_{\vk'}\rangle=(2\pi)^3\delta^3(\vk+\vk')P(k)$ \eqref{eq:fourier};
every quantity expanded as $X=\eps X_1+\tfrac{\eps^2}2X_2+\tfrac{\eps^3}6X_3$ \eqref{eq:expansion}; Poisson gauge with linear
metric functions \eqref{eq:poisson}; $a=\eta$.  $\nhat$ is the observation direction, the ray is $\vx=\vx_o+\nhat\chi$, and
$e^{+i\vk\cdot\nhat\chi}=4\pi\sum i^\ell\jl(k\chi)Y_{\ell m}(\nhat)Y^*_{\ell m}(\khat)$ \eqref{eq:rayleigh} ($Y_{\ell m}$ with Condon--Shortley phase).
Observable: emission isotropic in the fluid rest frame on $\Tphys=T_*$; observer = fluid element labelled at its $\Tphys=T_i$
event, fixed proper time, FW frame (equivalent to the canonical triad for $C_\ell$ and for $C^{(22)}_\ell$, $\Re C^{(13)}_\ell$ separately); $\Theta=\Tobs/\sky{\Tobs}-1$, dipole removed
after normalising (assumptions (O1)--(O3), Sec.~\ref{sec:setup}).  Monochromatic: $P=2\pi^2\sigma^2\delta(k-k_0)/k_0^2$ with
$\sigma^2=\langle\zeta_1^2\rangle$ held fixed, applied after assembly \eqref{eq:mono}.
\end{minipage}}

\paragraph{Main results.}
\begin{itemize}\itemsep1pt
\item[(1)] \emph{Loop formulas in named objects} (Secs.~\ref{sec:harmonic}, \ref{sec:assembly}):
$C^{(11)}_\ell=\frac2\pi\int k^2P\Delta_\ell^2$ (Prop.~\ref{prop:Delta}, ISW in closed form, Thm.~\ref{thm:isw});
$C^{(22)}_\ell=\frac1{4\pi^4(2\ell+1)}\int k_1^2k_2^2PP\int\dd\mu\sum_m|\tilde K_{2,\ell m}|^2$ \eqref{eq:asm-C22} with the symmetric kernel $K_2$ \eqref{eq:asm-K2};
$C^{(13)}_\ell=\frac2\pi\int k^2P\,\Delta_\ell\Delta^{\kappa*}_\ell$ \eqref{eq:asm-C13} with the contracted kernel $\kappa$ \eqref{eq:kappa-def}.
\item[(2)] \emph{Squeezed and IR theorems}: $K_2(\vk_1\to0,\vk_2)=-\vk_2\cdot\partial_{\vk_2}K_1$ (Thm.~\ref{thm:asm-squeezed}), $K_3(\vq,-\vq,\vk)=(\vk\cdot\partial_\vk)^2K_1+O(q^2)$
(Thm.~\ref{thm:asm-squeezed3}), both verified on the assembled kernels; hence the soft region of $\tfrac14C^{(22)}+\tfrac13\Re C^{(13)}$ is
$\tfrac12(n_s-1)^2\sigma_L^2C^{(11)}_\ell$ for a power law \eqref{eq:asm-IRns}: IR finite for $n_s\ge1$.
\item[(3)] \emph{Corners of the monochromatic loop}: the $(22)$ integrand is bounded at $K\to0$ for $\ell\ge2$ (Prop.~\ref{prop:mono-K0}); the collinear
poles of the $(13)$ integrand are a residual gauge mode and cancel identically (Prop.~\ref{prop:mono-residues}, Sec.~\ref{sec:mono-gauge}); the zero modes
are classified and the map is invariant under them (Sec.~\ref{sec:asm-zeromodes}).
\item[(4)] \emph{Final result} (monochromatic spectrum, boxed in \eqref{eq:mono-final}):
\[C_\ell=\eps^2\,4\pi\sigma^2\Delta_\ell^2+\eps^4\Big[\frac{\sigma^4}{4(2\ell+1)}\int\dd\mu\sum_m|\tilde K_{2,\ell m}|^2+\frac{4\pi}3\sigma^4\Delta_\ell\Re\Delta^\kappa_\ell\Big],\]
a function of $(k_0\eta_*,k_0\eta_o)$ at fixed $\sigma^2$, evaluated at $k_0\eta_*=2$, $k_0\eta_o=8$ (Table~\ref{tab:mono-comb}): in units of $\sigma^4$,
$\tfrac14C^{(22)}_\ell+\tfrac13\Re C^{(13)}_\ell=8.860$, $-7.198$, $3.862\times10^{-2}$ for $\ell=2,4,10$, against $C^{(11)}_\ell/\sigma^2=0.9546$, $0.1751$,
$9.68\times10^{-6}$; $C^{(22)}_\ell$ is confirmed by a Monte Carlo (Sec.~\ref{sec:mono-mc}); superhorizon, the $O(\eps^4)$ coefficient vanishes like
$C^{(11)}_\ell$ (Sec.~\ref{sec:mono-comb}).
\end{itemize}

\paragraph{Three structural results.}
\begin{itemize}\itemsep1pt
\item[(a)] Exact observable (Thm.~\ref{thm:exact}, Cor.~\ref{cor:theta}):\quad $\Tobs(\nhat)=T_o\,e^{-S(\nhat)}$,\quad $1+\Theta=e^{-S}/\sky{e^{-S}}$.
\item[(b)] Initial data (Def.~\ref{def:gauss}):\quad $\zeta_{\rm nl}=\eps\zeta_1$ Gaussian, $h_n(0)=0$, regular $\Rightarrow$ $X_n(\eta_i)$ fixed, non-local, constraint-satisfying.
\item[(c)] Log-normalisation algebra \eqref{eq:theta_orders}, with $m_n$ the coefficients of $e^{-S}$:
$\Theta^{(3)}_{\ell m}=m_{3,\ell m}-3\sky{m_1}m_{2,\ell m}-3\sky{m_2}m_{1,\ell m}+6\sky{m_1}^2m_{1,\ell m}$.
\end{itemize}
