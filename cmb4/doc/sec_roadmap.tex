\section{Roadmap: target, structure, status}
\label{sec:roadmap}

\noindent\fbox{\begin{minipage}{0.97\textwidth}\small
\textbf{Target and conventions.}
$C_\ell=\eps^2C^{(11)}_\ell+\eps^4[\tfrac14C^{(22)}_\ell+\tfrac13\Re C^{(13)}_\ell]+O(\eps^6)$, $\ell\ge2$ \eqref{eq:cl_target}, with
$C^{(rs)}_\ell=\frac1{2\ell+1}\sum_m\conn{a^{(r)}_{\ell m}a^{(s)*}_{\ell m}}$, $\conn{\cdot}$ connected in the $a_{\ell m}$ \eqref{eq:conn}.
Fourier $f=\int\frac{\dd^3k}{(2\pi)^3}e^{i\vk\cdot\vx}f_\vk$, $\langle\zeta_\vk\zeta_{\vk'}\rangle=(2\pi)^3\delta^3(\vk+\vk')P(k)$ \eqref{eq:fourier};
every quantity expanded as $X=\eps X_1+\tfrac{\eps^2}2X_2+\tfrac{\eps^3}6X_3$ \eqref{eq:expansion}; Poisson gauge with linear
metric functions \eqref{eq:poisson}; $a=\eta$.  $\nhat$ is the observation direction, the ray is $\vx=\vx_o+\nhat\chi$, and
$e^{+i\vk\cdot\nhat\chi}=4\pi\sum i^\ell\jl(k\chi)Y_{\ell m}(\nhat)Y^*_{\ell m}(\khat)$ \eqref{eq:rayleigh} ($Y_{\ell m}$ with Condon--Shortley phase).
Observable: emission isotropic in the fluid rest frame on $\Tphys=T_*$; observer = fluid element labelled at its $\Tphys=T_i$
event, fixed proper time, FW frame (replaceable by the canonical triad for $C_\ell$); $\Theta=\Tobs/\sky{\Tobs}-1$, dipole removed
after normalising (assumptions (O1)--(O3), Sec.~\ref{sec:setup}).  Monochromatic: $P=2\pi^2\sigma^2\delta(k-k_0)/k_0^2$ with
$\sigma^2=\langle\zeta_1^2\rangle$ held fixed, applied after assembly \eqref{eq:mono}.
\end{minipage}}

\paragraph{Three structural results.}
\begin{itemize}\itemsep1pt
\item[(a)] Exact observable (Thm.~\ref{thm:exact}, Cor.~\ref{cor:theta}):\quad $\Tobs(\nhat)=T_o\,e^{-S(\nhat)}$,\quad $1+\Theta=e^{-S}/\sky{e^{-S}}$.
\item[(b)] Initial data (Def.~\ref{def:gauss}):\quad $\zeta_{\rm nl}=\eps\zeta_1$ Gaussian, $h_n(0)=0$, regular $\Rightarrow$ $X_n(\eta_i)$ fixed, non-local, constraint-satisfying.
\item[(c)] Log-normalisation algebra \eqref{eq:theta_orders}, with $m_n$ the coefficients of $e^{-S}$:\quad
$\Theta^{(3)}_{\ell m}=m_{3,\ell m}-3\sky{m_1}m_{2,\ell m}-3\sky{m_2}m_{1,\ell m}+6\sky{m_1}^2m_{1,\ell m}$.
\end{itemize}

\paragraph{The $(13)$ term and the $O(\eps^4)$ coefficient.}  The contracted third-order kernel is assembled and validated
(Secs.~\ref{sec:asm-K3c}--\ref{sec:asm-IR}); the monochromatic $C^{(13)}_\ell$ is finite (numerical cancellation of the collinear residues,
Sec.~\ref{sec:mono-13}) and evaluated at $k_0=1$, $\eta_*=2$, $\eta_o=8$ (not at $\eta_o=4$).  In units of $\sigma^4$
(Table~\ref{tab:mono-comb}):
$\tfrac13\Re C^{(13)}_\ell=-5.398$, $-9.064$, $-1.463\times10^{-3}$ and
$\tfrac14C^{(22)}_\ell+\tfrac13\Re C^{(13)}_\ell=8.860$, $-7.198$, $3.862\times10^{-2}$ for $\ell=2,4,10$, against
$C^{(11)}_\ell/\sigma^2=0.9546$, $0.1751$, $9.68\times10^{-6}$.  These are evaluations by quadrature (relative accuracy of the $(13)$ part
$\le8\times10^{-5}$); open: an analytic proof of the residue cancellation and the UV behaviour for power-law spectra (Sec.~\ref{sec:unresolved}).

\paragraph{Delicate points.}
\begin{itemize}\itemsep1pt
\item \emph{Dilation and squeezed limits.}  Constant and gradient modes of $\zeta_1$ must be unobservable; the exact dilation family is the $k=0$
gauge test (Sec.~\ref{sec:check_dilation}); the squeezed limit of $K_2$ is a dilation of $K_1$, $K_2(\vk_1\to0,\vk_2)=-\vk\cdot\partial_\vk K_1+O(k_1)$
(Sec.~\ref{sec:asm-squeezed}), and the soft region of the loop is a total derivative that vanishes only for $P\propto k^{-3}$.
\item \emph{Zero modes.}  The $k\to0$ limit of the second-order kernels does not exist \eqref{eq:KPhi20_smallk}; the tadpole convention averages
over angles first.  In the assembled kernel only the angle-averaged tadpole can enter, and the remaining freedom (a residual time
reparametrisation) drops out of $\kappa$ to $10^{-13}$ (Sec.~\ref{sec:asm-C13}); an analytic proof for general momenta is open.
\item \emph{$\mu\to\pm1$.}  In $(22)$, $K\to0$ is integrable for $\ell\ge2$ (Sec.~\ref{sec:mono-K0}).  In $(13)$ at $q=k$, $C_3$ has a simple pole in $\mu$
\eqref{eq:C3pole}; six channels of $\kappa$ have $1/(1-\mu)$ poles whose residues cancel numerically, so the integrand has a finite limit
(Sec.~\ref{sec:mono-13}); an analytic residue computation is open.
\item \emph{UV behaviour of the perfect fluid.}  After emission the fluid is continued as a perfect fluid (no free streaming); the acoustic
oscillations persist undamped, $\Delta_\ell\to$ oscillatory as $k\to\infty$ (Sec.~\ref{sec:linear}), and the $(22)$ kernel reaches $K=2k_0$, so
$\tfrac14C^{(22)}_\ell/C^{(11)}_\ell$ grows steeply with $\ell$ for a monochromatic spectrum (Sec.~\ref{sec:mono-num}): the expansion in
$\eps^2\sigma^2$ is $\ell$ dependent.
\item \emph{IR for tilted spectra.}  The IR region of the one-loop terms cancels between $\tfrac14C^{(22)}$ and $\tfrac13\Re C^{(13)}$ only up to a
total derivative $\propto(3+\vk\cdot\partial_\vk)^2P$; for $n_s\neq1$ the loop is IR sensitive (Sec.~\ref{sec:asm-IR}).  The monochromatic
spectrum has no IR region.
\end{itemize}
